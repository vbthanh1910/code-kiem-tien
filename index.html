const express = require('express');
const cors = require('cors');

const app = express();
const PORT = 3000;

// Cho phép truyền dữ liệu CORS chéo tên miền
app.use(cors());
app.use(express.json());

// Endpoint tiếp nhận dữ liệu tư vấn dinh dưỡng
app.post('/api/receive-nutrition-data', (req, res) => {
    const payload = req.body;
    
    console.log('=== [SERVER MÁY NHẬN] ĐÃ NHẬN DỮ LIỆU ===');
    console.log('Thời gian:', payload.clientTimestamp);
    console.log('Thông tin người dùng:', payload.userData);
    console.log('Chỉ số tính toán:', payload.calculatedMetrics);
    console.log('Macros (Gram):', payload.macros);

    res.status(200).json({
        status: 'success',
        message: 'Server máy nhận đã xử lý và lưu dữ liệu thành công!',
        receivedAt: new Date().toISOString()
    });
});

app.listen(PORT, '0.0.0.0', () => {
    console.log(`Server đang lắng nghe tại: http://localhost:${PORT}`);
});


                
                
from flask import Flask, request, jsonify
from flask_cors import CORS
from datetime import datetime

app = Flask(__name__)
CORS(app)

@app.route('/api/receive-nutrition-data', methods=['POST'])
def receive_nutrition_data():
    data = request.get_json()
    
    print("=== [SERVER PYTHON] DỮ LIỆU ĐÃ TỚI ===")
    print("User Data:", data.get('userData'))
    print("Metrics:", data.get('calculatedMetrics'))
    
    return jsonify({
        "status": "success",
        "message": "Python Server đã nhận kết quả thành công!",
        "receivedAt": datetime.now().isoformat()
    }), 200

if __name__ == '__main__':
    app.run(host='0.0.0.0', port=3000)
